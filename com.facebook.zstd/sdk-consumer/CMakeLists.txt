cmake_minimum_required(VERSION 3.20)
project(ProjectedZstdSdk C)
find_package(zstd CONFIG REQUIRED)
enable_testing()
foreach(mode IN ITEMS shared static)
  add_executable(consumer-${mode} consumer.c)
  target_link_libraries(consumer-${mode} PRIVATE zstd::libzstd_${mode})
  add_test(NAME zstd-${mode}-roundtrip COMMAND consumer-${mode})
endforeach()
