cmake_minimum_required(VERSION 3.20)
project(ProjectedCapstoneSdk C)
find_package(capstone CONFIG REQUIRED)
enable_testing()
foreach(mode IN ITEMS shared static)
  add_executable(consumer-${mode} consumer.c)
  target_link_libraries(consumer-${mode} PRIVATE capstone::capstone_${mode})
  add_test(NAME capstone-${mode}-decode COMMAND consumer-${mode})
endforeach()
add_test(NAME capstone-imported-cstool COMMAND capstone::cstool x64 554889e5)
set_tests_properties(capstone-imported-cstool PROPERTIES PASS_REGULAR_EXPRESSION "push")
