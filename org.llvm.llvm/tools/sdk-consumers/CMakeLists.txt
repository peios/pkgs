cmake_minimum_required(VERSION 3.20)
project(PeiosClangLldSdkConsumers LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(LLVM REQUIRED CONFIG)
include(CTest)
separate_arguments(LLVM_DEFINITION_FLAGS NATIVE_COMMAND "${LLVM_DEFINITIONS}")

if(SDK_FAMILY STREQUAL "llvm")
  foreach(mode IN ITEMS shared static)
    add_executable(llvm-${mode}-consumer llvm-consumer.cpp)
    target_include_directories(llvm-${mode}-consumer PRIVATE ${LLVM_INCLUDE_DIRS})
    target_compile_options(llvm-${mode}-consumer PRIVATE ${LLVM_DEFINITION_FLAGS})
    if(mode STREQUAL "shared")
      target_link_libraries(llvm-${mode}-consumer PRIVATE LLVM)
    else()
      target_link_libraries(llvm-${mode}-consumer PRIVATE LLVMCore LLVMSupport)
    endif()
    add_test(NAME llvm-${mode}-ir COMMAND llvm-${mode}-consumer)
  endforeach()
elseif(SDK_FAMILY STREQUAL "clang")
  find_package(Clang REQUIRED CONFIG)
foreach(mode IN ITEMS shared static)
  add_executable(clang-${mode}-consumer clang-consumer.cpp)
  target_include_directories(clang-${mode}-consumer PRIVATE ${CLANG_INCLUDE_DIRS} ${LLVM_INCLUDE_DIRS})
  target_compile_options(clang-${mode}-consumer PRIVATE ${LLVM_DEFINITION_FLAGS})
  if(mode STREQUAL "shared")
    target_link_libraries(clang-${mode}-consumer PRIVATE clang-cpp LLVM)
  else()
    target_link_libraries(clang-${mode}-consumer PRIVATE clangTooling)
  endif()
  add_test(NAME clang-${mode}-parse COMMAND clang-${mode}-consumer)
endforeach()

elseif(SDK_FAMILY STREQUAL "lld")
  find_package(LLD REQUIRED CONFIG)
add_library(sdk-link-entry OBJECT entry.c)
set_target_properties(sdk-link-entry PROPERTIES POSITION_INDEPENDENT_CODE OFF)
target_compile_options(sdk-link-entry PRIVATE -ffreestanding -fno-pie)
add_executable(lld-static-consumer lld-consumer.cpp)
target_include_directories(lld-static-consumer PRIVATE ${LLD_INCLUDE_DIRS} ${LLVM_INCLUDE_DIRS})
target_compile_options(lld-static-consumer PRIVATE ${LLVM_DEFINITION_FLAGS})
target_link_libraries(lld-static-consumer PRIVATE lldELF)
add_dependencies(lld-static-consumer sdk-link-entry)
add_test(NAME lld-static-link COMMAND lld-static-consumer $<TARGET_OBJECTS:sdk-link-entry> ${CMAKE_CURRENT_BINARY_DIR}/sdk-linked-elf)
set_tests_properties(lld-static-link PROPERTIES FIXTURES_SETUP lld-linked-elf)
add_test(NAME lld-linked-elf-exec COMMAND ${CMAKE_CURRENT_BINARY_DIR}/sdk-linked-elf)
set_tests_properties(lld-linked-elf-exec PROPERTIES FIXTURES_REQUIRED lld-linked-elf)
else()
  message(FATAL_ERROR "Unknown SDK family: ${SDK_FAMILY}")
endif()
